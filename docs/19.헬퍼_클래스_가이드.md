        $patterns = [
            "user_{$userId}_*",
            "user_posts_{$userId}",
            "user_profile_{$userId}"

        foreach ($patterns as $pattern) {
            CacheHelper::deletePattern($pattern);

            "post_{$postId}",
            "post_comments_{$postId}",

    public static function processInChunks($data, $chunkSize = 1000, $callback) {
        $chunks = array_chunk($data, $chunkSize);
        
        foreach ($chunks as $chunk) {
            $callback($chunk);
            
            // 메모리 정리
            if (memory_get_usage() > 100 * 1024 * 1024) { // 100MB

            }

    }
}

    }
);

#### 6.4 FCM 푸시 알림 통합 사례 (v3.89.0)
```php
// CommentController.php - 댓글 작성 시 푸시 전송
try {
    // 1. 댓글 저장
    $comment = Comment::create($data);

    // 2. 게시글 작성자의 알림 설정 확인
    if (NotificationSettings::isNotificationEnabled($postAuthorId, 'comments')) {
        // 3. FCM 토큰 조회
        $tokens = FcmToken::getActiveTokensByUserId($postAuthorId);

        // 4. 푸시 전송
        if (!empty($tokens)) {
            $result = FcmHelper::sendBulkPush(
                $tokens,
                '새로운 댓글',
                $comment['content'],
                [
                    'post_id' => $postId,
                    'comment_id' => $comment['id'],
                    'type' => 'comment'
                ]
            );

            WebLogger::info('댓글 알림 전송 완료', $result);
        }
    } else {
        WebLogger::info('댓글 알림 스킵', ['user_id' => $postAuthorId]);
    }

} catch (Exception $e) {
    // 푸시 실패해도 댓글 저장은 성공 (Core 기능 보장)
    WebLogger::error('FCM 푸시 전송 실패', ['error' => $e->getMessage()]);
}
```

#### 6.5 FCM 푸시 알림 9가지 시나리오 (v3.89.0)
1. **댓글 알림** - 내 게시글에 댓글 작성 시 → CommentController
