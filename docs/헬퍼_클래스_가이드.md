class CacheHelper {
    // 캐시 저장
    public static function set($key, $value, $ttl = 3600)
    
    // 캐시 조회
    public static function get($key, $default = null)
    
    // 캐시 삭제
    public static function delete($key)
    
    // 캐시 존재 확인
    public static function exists($key)
    
    // 패턴으로 캐시 삭제
    public static function deletePattern($pattern)
    
    // Remember 패턴 (없으면 생성, 있으면 반환)
    public static function remember($key, $ttl, $callback)
}

// 사용 예시
// 기본 캐싱
CacheHelper::set('user_123_profile', $userProfile, 1800); // 30분
$profile = CacheHelper::get('user_123_profile');

// Remember 패턴
$popularPosts = CacheHelper::remember(
    'popular_posts_today',
    3600, // 1시간
    function() {
        return Post::getPopularToday();
    }
);
```

// 계층적 캐시 무효화
class CacheInvalidation {
    public static function invalidateUser($userId) {
        $patterns = [
            "user_{$userId}_*",
            "user_posts_{$userId}",
            "user_profile_{$userId}"
        ];
        
        foreach ($patterns as $pattern) {
            CacheHelper::deletePattern($pattern);
        }
    }
    
    public static function invalidatePost($postId) {
        $patterns = [
            "post_{$postId}",
            "post_comments_{$postId}",
            "popular_posts_*",
            "recent_posts_*"
        ];
        
        foreach ($patterns as $pattern) {
            CacheHelper::deletePattern($pattern);
        }
    }
}
```

class ContentChunker {
    public static function processInChunks($data, $chunkSize = 1000, $callback) {
        $chunks = array_chunk($data, $chunkSize);
        
        foreach ($chunks as $chunk) {
            $callback($chunk);
            
            // 메모리 정리
            if (memory_get_usage() > 100 * 1024 * 1024) { // 100MB
                gc_collect_cycles();
            }
        }
    }
}

        }
    }
);

#### 9.2 Service에서의 활용
```php
class LectureService {
    public function createLecture($data) {
        // 성능 측정 시작
        PerformanceDebugger::start('lecture_creation');
        
        try {
            // 데이터 검증
            $this->validateLectureData($data);
            
            // 강의 생성
            $lecture = Lecture::create($data);
            
            // 캐시 무효화
            CacheHelper::deletePattern('lectures_*');
            
            // SMS 알림 발송
            SmsHelper::sendLectureCreatedSms($data['instructor_phone'], $lecture->title);
            
            // 로깅
            WebLogger::info('강의 생성 완료', [
                'lecture_id' => $lecture->id,
                'title' => $lecture->title
            ]);
            
            return $lecture;
            
        } catch (Exception $e) {
            WebLogger::error('강의 생성 실패', [
                'error' => $e->getMessage(),
                'data' => $data
            ]);
            throw $e;
        } finally {
            PerformanceDebugger::end('lecture_creation');
        }
    }
}
```

// 1. 정적 메서드 사용
class UtilityHelper {
    public static function formatCurrency($amount) {
        return number_format($amount) . '원';
    }
}

// 2. 에러 처리 포함
class ApiHelper {
    public static function makeRequest($url, $data = []) {
        try {
            $response = curl_exec($curl);
            if ($response === false) {
                throw new Exception('API 요청 실패: ' . curl_error($curl));
            }
            return json_decode($response, true);
        } catch (Exception $e) {
            WebLogger::error('API 요청 오류', [
                'url' => $url,
                'error' => $e->getMessage()
            ]);
            return false;
        }
    }
}

// 3. 캐싱 통합
class DataHelper {
    public static function getExpensiveData($id) {
        return CacheHelper::remember(
            "expensive_data_{$id}",
            3600,
            function() use ($id) {
                return self::fetchExpensiveData($id);
            }
        );
    }
}

// 4. 로깅 통합
class ProcessHelper {
    public static function processItems($items) {
        WebLogger::info('배치 처리 시작', ['count' => count($items)]);
        
        $processed = 0;
        foreach ($items as $item) {
            try {
                self::processItem($item);
                $processed++;
            } catch (Exception $e) {
                WebLogger::error('아이템 처리 실패', [
                    'item_id' => $item['id'],
                    'error' => $e->getMessage()
                ]);
            }
        }
        
        WebLogger::info('배치 처리 완료', [
            'total' => count($items),
            'processed' => $processed,
            'failed' => count($items) - $processed
        ]);
        
        return $processed;
    }
}
